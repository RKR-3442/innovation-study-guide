<script>
// Function to toggle question visibility
function toggleQuestion(header) {
    header.classList.toggle('active');
    const content = header.nextElementSibling;
    content.classList.toggle('show');
}

// Function to toggle answer visibility
function toggleAnswer(button) {
    button.classList.toggle('active');
    const answer = button.nextElementSibling;
    answer.classList.toggle('show');
    
    if (button.classList.contains('active')) {
        button.textContent = '▲ Hide Answer';
    } else {
        button.textContent = '▼ Show Model Answer';
    }
}

// Function to switch viewing modes
function setMode(mode) {
    const body = document.body;
    const buttons = document.querySelectorAll('.mode-btn');
    
    // Remove all mode classes
    body.classList.remove('study-mode', 'practice-mode', 'quick-mode');
    
    // Add selected mode
    body.classList.add(mode + '-mode');
    
    // Update button states
    buttons.forEach(btn => btn.classList.remove('active'));
    document.getElementById(mode + 'Mode').classList.add('active');
    
    // Log for debugging
    console.log('Mode switched to:', mode);
}

// Initialize in study mode
document.addEventListener('DOMContentLoaded', function() {
    setMode('study');
});
</script>
